### 3. 各组件实现要点

#### 3.1 Harness 类（`harness.py`）

这是 notes.md 里 `build_agent()` 返回的那个 `H`。它的职责是「把所有零件拼到一起」：

```python
import yaml
from dataclasses import dataclass
from typing import Any

@dataclass
class Harness:
    """Harness = 所有工程零件的容器。LLM 不在这里面。"""
    system_prompt: str
    rules: str                          # 拼接后的规则文本
    skills: "SkillRegistry"
    tools: dict[str, "Tool"]
    mcp: "MCPHub"
    hooks: "HookRunner"
    guardrail: "Guardrail"
    sandbox: "Sandbox"
    tracer: "Tracer"
    memory: "MemoryStore"
    retriever: "Retriever"
    config: dict                        # 原始配置，给子 Agent 用

    @classmethod
    def from_yaml(cls, path: str) -> "Harness":
        """从 agent.yaml 装配整个 Harness。这就是 build_agent() 的实现。"""
        with open(path) as f:
            cfg = yaml.safe_load(f)

        return cls(
            system_prompt=cls._build_system_prompt(cfg),
            rules=cls._load_rules(cfg),
            skills=SkillRegistry.from_yaml(cfg["skills"]["index"]),
            tools=ToolLoader.load_all(cfg["tools"]["builtin"]),
            mcp=MCPHub.from_config(cfg["mcp"]),
            hooks=HookRunner.from_config(cfg.get("hooks", {})),
            guardrail=Guardrail.from_yaml(cfg["guardrail"]["policy"]),
            sandbox=SandboxFactory.create(cfg["sandbox"]),
            tracer=Tracer.from_config(cfg["tracer"]),
            memory=MemoryStore.from_config(cfg["memory"]),
            retriever=Retriever.from_config(cfg["retriever"]),
            config=cfg,
        )

    def restrict(self, scope: dict) -> "Harness":
        """为子 Agent 生成受限版本的 Harness。
        对应 notes.md 里的 H.restrict(action.scope)。"""
        # 关键：工具子集 + 独立沙箱 + 更小的 token 预算
        restricted_cfg = deepcopy(self.config)
        restricted_cfg["agent"]["max_steps"] = scope.get("max_steps", 10)
        restricted_cfg["tools"]["builtin"] = [
            t for t in restricted_cfg["tools"]["builtin"]
            if t["name"] in scope.get("allowed_tools", [])
        ]
        return Harness.from_config(restricted_cfg)  # 递归构建

    # ... 省略 _build_system_prompt, _load_rules 等辅助方法
```

#### 3.2 工具原语（`tools/`）

notes.md 说工具是「读写文件 / 跑 bash，agent 最底层的动作」。每个工具是一个标准的 Python 对象：

```python
# tools/file_tools.py
from dataclasses import dataclass
from typing import Any

@dataclass
class Tool:
    """工具的标准接口。注册到 Harness.tools 的每个条目都是这个类型。"""
    name: str
    description: str                   # ← 这是 LLM 看到的东西，决定何时调用
    parameters: dict                   # JSON Schema
    handler: callable                  # 实际执行的函数
    sandbox: bool = True               # 是否需要在沙箱内执行
    require_approval: bool = False     # 是否需要 guardrail 放行

    def to_openai_schema(self) -> dict:
        """转换为 OpenAI Function Calling 格式。"""
        return {
            "type": "function",
            "function": {
                "name": self.name,
                "description": self.description,
                "parameters": self.parameters,
            }
        }

# 示例：注册一个工具
read_file_tool = Tool(
    name="read_file",
    description="读取指定路径的文件内容。仅在需要查看文件时使用。",
    parameters={
        "type": "object",
        "properties": {
            "path": {
                "type": "string",
                "description": "相对于项目根目录的文件路径"
            }
        },
        "required": ["path"]
    },
    handler=lambda path: open(path).read(),
    sandbox=True,
    require_approval=False,            # 只读操作无需审批
)
```

#### 3.3 权限门控（`guardrails/policy.yaml`）

notes.md 里 `H.guardrail.allow(action)` 的具体实现：

```yaml
# guardrails/policy.yaml
# 门控策略：决定什么动作可以自动执行，什么需要人工审批，什么直接拒绝

rules:
  # 规则按优先级从高到低匹配

  - pattern: "rm -rf /*"               # 黑名单：直接拒绝
    action: deny
    reason: "禁止删除根目录"

  - pattern: "rm *"
    action: ask_user                   # 灰名单：询问用户
    reason: "删除文件需要确认"

  - tools: ["write_file", "run_shell", "http_request"]
    scope: "outside_workspace"         # 在项目目录外操作
    action: ask_user
    reason: "在项目目录外操作需要确认"

  - tools: ["http_request"]
    method: ["POST", "PUT", "DELETE"]  # 写操作
    action: ask_user
    reason: "对外写请求需要确认"

  - tools: ["read_file", "web_search"]
    action: allow                      # 白名单：自动放行
```

```python
# guardrails/guardrail.py
class Guardrail:
    def __init__(self, rules: list[dict]):
        self.rules = rules

    def allow(self, action) -> tuple[bool, str]:
        """返回 (是否放行, 原因)。不放行时把原因回灌给 LLM。"""
        for rule in self.rules:
            if self._matches(action, rule):
                verdict = rule["action"]
                if verdict == "deny":
                    return False, f"动作被拒绝: {rule['reason']}"
                elif verdict == "ask_user":
                    # 暂停循环，等用户输入 y/n
                    user_input = input(f"⚠️ {rule['reason']}。允许吗? (y/n): ")
                    return user_input.lower() == "y", f"用户{'批准' if user_input.lower() == 'y' else '拒绝'}"
                else:  # allow
                    return True, ""
        return True, ""  # 默认放行
```

#### 3.4 Hooks（`hooks/`）

Hooks 和 Guardrail 是「同一套闸的两种用法」。Hooks 不拦截，而是**触发副作用**。

```python
# hooks/pre_tool_use.py
def pre_tool_use(action, context: dict) -> dict | None:
    """
    PreToolUse 钩子。在工具执行前触发。
    可以返回 {"action": "deny", "reason": "..."} 来拦截。
    可以返回 {"action": "modify", "args": {...}} 来修改参数。
    返回 None 表示放行。
    """
    if action.tool == "run_shell":
        # 记录所有 shell 命令到审计日志
        with open("/var/log/agent-shell-audit.log", "a") as f:
            f.write(f"[{datetime.now().isoformat()}] {action.args}\n")

    # 示例：自动替换危险的 shell 参数
    if action.tool == "run_shell" and "sudo" in str(action.args):
        return {"action": "modify", "args": str(action.args).replace("sudo", "")}

    return None  # 放行
```

#### 3.5 沙箱（`sandbox/docker_sandbox.py`）

notes.md 里 `H.sandbox.run()` 的实现。核心诉求是「隔离 + 资源限制 + 超时」：

```python
import subprocess
import tempfile
import os

class DockerSandbox:
    """用 Docker 容器做沙箱隔离。"""
    
    def __init__(self, image: str, memory_limit: str, cpu_limit: str, timeout: int, network: str):
        self.image = image
        self.memory_limit = memory_limit
        self.cpu_limit = cpu_limit
        self.timeout = timeout
        self.network = network

    def run(self, tool_name: str, args: dict) -> str:
        """在隔离容器中执行工具调用。"""
        
        # 写一个临时脚本
        with tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False) as f:
            f.write(f"""
import json, sys
from tools.{tool_name} import handler
try:
    result = handler(**{json.dumps(args)})
    print(json.dumps({{"ok": True, "result": result}}))
except Exception as e:
    print(json.dumps({{"ok": False, "error": str(e)}}))
""")
            script_path = f.name

        try:
            proc = subprocess.run([
                "docker", "run", "--rm",
                f"--memory={self.memory_limit}",
                f"--cpus={self.cpu_limit}",
                f"--network={self.network}",
                "-v", f"{script_path}:/script.py:ro",
                "-v", f"{os.getcwd()}:/workspace:rw",  # 只挂载工作区
                self.image,
                "python", "/script.py"
            ], capture_output=True, text=True, timeout=self.timeout)
            
            result = json.loads(proc.stdout)
            if result["ok"]:
                return result["result"]
            else:
                raise ToolError(result["error"])
        except subprocess.TimeoutExpired:
            raise ToolError(f"工具 {tool_name} 执行超时 ({self.timeout}s)")
        finally:
            os.unlink(script_path)


class LocalSandbox:
    """开发环境用：本地进程沙箱（不安全，仅用于快速迭代）。"""
    
    def run(self, tool_name: str, args: dict) -> str:
        """直接在本地执行（无隔离，仅开发用）。"""
        handler = TOOL_REGISTRY[tool_name]
        return handler(**args)


class SandboxFactory:
    @staticmethod
    def create(config: dict) -> "Sandbox":
        if config["type"] == "docker":
            return DockerSandbox(**config)
        elif config["type"] == "local":
            return LocalSandbox()
        else:
            raise ValueError(f"未知沙箱类型: {config['type']}")
```

#### 3.6 MCP 连接（`mcp/`）

notes.md 说 MCP 是「把外部工具按统一协议接进来，N+M，不为每个工具改代码」。实践中的连接管理：

```python
# mcp/mcp_hub.py
import asyncio
import subprocess
import json

class MCPHub:
    """
    MCP 连接管理器。
    管理多个 MCP Server 的生命周期，并把它们的工具统一暴露为菜单。
    """
    
    def __init__(self, server_configs: list[dict]):
        self.servers: dict[str, MCPServerConnection] = {}
        self._tools_cache: list[dict] | None = None
        for cfg in server_configs:
            self.servers[cfg["name"]] = MCPServerConnection(cfg)

    async def start_all(self):
        """启动所有 MCP Server（在 build_agent 阶段调用）。"""
        for name, conn in self.servers.items():
            await conn.start()
        # 启动后刷新工具列表
        await self._refresh_tools()

    async def _refresh_tools(self):
        """从所有已连接的 Server 拉取工具列表。"""
        all_tools = []
        for name, conn in self.servers.items():
            tools = await conn.list_tools()
            for tool in tools:
                tool["_mcp_server"] = name  # 标记来源
                all_tools.append(tool)
        self._tools_cache = all_tools

    def list_tools(self) -> list[dict]:
        """返回所有 MCP 工具（给 LLM 看的功能菜单的一部分）。"""
        return self._tools_cache or []

    def call(self, tool_name: str, args: dict) -> str:
        """调用 MCP 工具（在外部执行，本地沙箱覆盖不到）。"""
        for conn in self.servers.values():
            for tool in conn.tools:
                if tool["name"] == tool_name:
                    return conn.call_tool(tool_name, args)
        raise ToolError(f"MCP 工具不存在: {tool_name}")

    async def close_all(self):
        """关闭所有连接（SessionEnd 时调用）。"""
        for conn in self.servers.values():
            await conn.close()


class MCPServerConnection:
    """单个 MCP Server 的连接管理。"""
    
    def __init__(self, config: dict):
        self.name = config["name"]
        self.command = config["command"]
        self.args = config.get("args", [])
        self.env = {**os.environ, **config.get("env", {})}
        self.process: subprocess.Popen | None = None
        self.tools: list[dict] = []

    async def start(self):
        """启动 MCP Server 子进程，建立 stdio 通信。"""
        self.process = subprocess.Popen(
            [self.command] + self.args,
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            env=self.env,
        )
        # 发送 initialize 请求...
        await self._send_request("initialize", {"protocolVersion": "2024-11-05"})
        # 拉取工具列表...
        resp = await self._send_request("tools/list", {})
        self.tools = resp["tools"]

    async def call_tool(self, name: str, args: dict) -> str:
        resp = await self._send_request("tools/call", {"name": name, "arguments": args})
        return json.dumps(resp["content"])
```

**skills/code-review/SKILL.md**（命中后才加载的全文）：
````markdown
# Code Review Skill

## 触发条件
当用户要求 review 代码、检查 PR、或自审提交时，激活此技能。

## 执行步骤
1. 用 `git diff` 获取所有未提交或指定范围的变更
2. 对每个变更文件，检查以下维度：
   - **正确性**：逻辑是否有误
   - **安全性**：是否有注入、泄露风险
   - **可读性**：命名是否清晰、函数是否过长（>50行）
   - **性能**：是否有 N+1 查询、不必要的循环
3. 输出格式（Markdown）：
```
## 评审报告
### 总体评价
### 严重问题 🔴
- [文件名:行号] 问题描述 + 建议
### 改进建议 🟡
- [文件名:行号] 问题描述 + 建议
### 亮点 🟢
- 值得肯定的写法
```

## 注意事项
- 不评审第三方依赖代码
- 安全问题的优先级高于风格问题
- 评审意见要给出具体修改建议，而非泛泛而谈
````

```python
# skills/registry.py
class SkillRegistry:
    """技能注册表：管理名片和全文的渐进式加载。"""
    
    def __init__(self, skills: list[dict]):
        self._skills = {s["name"]: s for s in skills}

    def list_descriptions(self) -> list[dict]:
        """返回所有技能的名片（仅 description，不含全文）。
        这是 LLM 菜单的一部分。"""
        return [
            {"name": name, "description": s["description"]}
            for name, s in self._skills.items()
        ]

    def load(self, name: str) -> str:
        """命中后才加载 SKILL.md 全文。
        返回的文本作为 user 消息注入上下文。"""
        skill = self._skills.get(name)
        if not skill:
            return f"技能 '{name}' 不存在。可用的技能: {list(self._skills.keys())}"
        with open(skill["path"]) as f:
            return f.read()

    @classmethod
    def from_yaml(cls, path: str) -> "SkillRegistry":
        with open(path) as f:
            data = yaml.safe_load(f)
        return cls(data["skills"])
```

```python
# tracer/tracer.py
import json
import time

class Tracer:
    """记录每一步决策和观察，事后可回放、可审计。"""

    def __init__(self, output_path: str):
        self.output_path = output_path
        self.trace_id = f"trace-{int(time.time())}"
        self.records: list[dict] = []
        self.start_time = time.time()

    def record(self, step: int, text: str, action, observation: str = None):
        """每轮循环记录一次。对应 notes.md 里的 H.tracer.record()。"""
        self.records.append({
            "trace_id": self.trace_id,
            "step": step,
            "timestamp": time.time() - self.start_time,
            "text": text,                      # LLM 说了什么
            "action": {                        # LLM 决定做什么
                "type": action.type,
                "tool": getattr(action, "tool", None),
                "args": getattr(action, "args", None),
            },
            "observation": observation,         # 工具返回了什么
        })

    def flush(self):
        """会话结束时落盘。对应 notes.md 里的 H.tracer.flush()。"""
        path = f"{self.output_path}/{self.trace_id}.jsonl"
        with open(path, "w") as f:
            for record in self.records:
                f.write(json.dumps(record, ensure_ascii=False) + "\n")
        print(f"📊 轨迹已保存: {path} ({len(self.records)} 步)")

    @classmethod
    def from_config(cls, config: dict) -> "Tracer":
        if config["type"] == "file":
            return cls(output_path=config["output"])
        elif config["type"] == "otel":
            return OtelTracer(config)  # 对接 OpenTelemetry
        elif config["type"] == "langfuse":
            return LangfuseTracer(config)
        raise ValueError(f"Unknown tracer type: {config['type']}")
```

### 4. 核心循环的完整实现（`agent_loop.py`）

把 notes.md 的 Step 6 伪代码翻译成可运行的 Python：

```python
import time
from typing import Any

class AgentLoop:
    """Agent 核心循环。这就是 notes.md 里那个 while 循环的完整实现。"""

    def __init__(self, harness: Harness, llm_client: Any):
        self.H = harness
        self.llm = llm_client

    def run(self, goal: str, depth: int = 0) -> str:
        H = self.H

        # ===== ① 上下文装配（开局一次性）=====
        context = []
        context.append({"role": "system", "content": H.system_prompt})
        context.append({"role": "system", "content": H.rules})
        
        memory_context = H.memory.read(goal)
        if memory_context:
            context.append({"role": "system", "content": memory_context})
        
        rag_context = H.retriever.retrieve(goal)
        if rag_context:
            context.append({"role": "system", "content": rag_context})
        
        context.append({"role": "user", "content": goal})

        # ===== ② 主循环 =====
        done = False
        answer = None
        steps = 0
        max_steps = H.config["agent"]["max_steps"]

        while not done and steps < max_steps:
            steps += 1
            H.tracer.record(steps, text="", action=None)  # 先占位

            # 熵管理：窗口将满时压缩
            if self._token_count(context) > H.config["agent"]["token_limit"]:
                context = self._compact(context)

            # 组装菜单：内建工具 + MCP 工具 + Skill 名片
            menu = (
                [t.to_openai_schema() for t in H.tools.values()] +
                H.mcp.list_tools() +
                H.skills.list_descriptions()
            )

            # ★ 唯一的 LLM 调用：任务决策
            response = self.llm.chat(context, tools=menu)

            text = response.text
            action = response.action  # 解析出来的动作对象

            # 将决策写入上下文
            context.append({"role": "assistant", "content": text, "action": action})
            H.tracer.record(steps, text=text, action=action)

            # --- 动作分发 ---

            if action.type == "done":
                done = True
                answer = text
                continue

            # 权限门控
            allowed, reason = H.guardrail.allow(action)
            if not allowed:
                context.append({"role": "user", "content": f"动作被拦截: {reason}"})
                H.tracer.record(steps, text=text, action=action, observation=f"GUARDRAIL_DENY: {reason}")
                continue

            # PreToolUse 钩子
            verdict = H.hooks.run("pre_tool_use", action)
            if verdict == "deny":
                context.append({"role": "user", "content": "动作被 Hook 拦截"})
                continue

            # --- 按动作类型分发 ---
            try:
                if action.type == "use_skill":
                    instructions = H.skills.load(action.name)
                    context.append({"role": "user", "content": instructions})

                elif action.type == "call_tool":
                    if action.tool in H.mcp.list_tool_names():
                        result = H.mcp.call(action.tool, action.args)
                    else:
                        result = H.sandbox.run(action.tool, action.args)
                    context.append({"role": "user", "content": result})
                    H.tracer.record(steps, text=text, action=action, observation=result)

                elif action.type == "spawn_subagent":
                    if depth >= H.config["agent"]["max_depth"]:
                        context.append({"role": "user", "content": "子 Agent 嵌套已达深度上限"})
                        continue
                    child_H = H.restrict(action.scope)
                    child_loop = AgentLoop(child_H, self.llm)
                    summary = child_loop.run(action.subtask, depth=depth + 1)
                    context.append({"role": "user", "content": f"[子 Agent 完成] {summary}"})

                elif action.type == "take_note":
                    H.memory.write(action.note)

                # 反馈循环：改了代码就跑传感器
                if action.changed_code:
                    report = self._run_sensors()
                    if report:
                        context.append({"role": "user", "content": report})

            except ToolError as e:
                context.append({"role": "user", "content": f"工具失败: {e}"})

            # PostToolUse 钩子
            H.hooks.run("post_tool_use", action)

        # ===== ③ 收尾 =====
        H.hooks.run("session_end", None)
        H.memory.consolidate(context)
        H.tracer.flush()

        return answer or "任务未完成（达到步数上限）"

    def _token_count(self, context: list) -> int:
        """估算 token 数。简单实现：字符数 / 2.5。"""
        total = sum(len(str(m.get("content", ""))) for m in context)
        return int(total / 2.5)

    def _compact(self, context: list) -> list:
        """压缩上下文（又是一次 LLM 调用）。"""
        # 保留 system 消息 + 最后 20 条 + LLM 压缩中间部分
        system_msgs = [m for m in context if m["role"] == "system"]
        recent = context[-20:]
        middle = context[len(system_msgs):-20]
        if middle:
            summary = self.llm.chat("请用 3 句话总结以下对话的关键信息:", 
                                     messages=middle)
            system_msgs.append({"role": "system", "content": f"[压缩的历史] {summary}"})
        return system_msgs + recent

    def _run_sensors(self) -> str:
        """反馈循环：跑测试/Lint/类型检查。"""
        reports = []
        # 跑 lint
        lint_result = subprocess.run(["ruff", "check", "."], capture_output=True, text=True)
        if lint_result.stdout:
            reports.append(f"## Lint 结果\n```\n{lint_result.stdout[:2000]}\n```")
        # 跑测试
        test_result = subprocess.run(["pytest", "-q"], capture_output=True, text=True)
        if test_result.returncode != 0:
            reports.append(f"## 测试失败\n```\n{test_result.stdout[:2000]}\n```")
        return "\n\n".join(reports)
```

```python
# multi_agent_patterns.py

# ===== 模式 1: Pipeline =====
async def pipeline(harness: Harness, llm, task: str):
    """研究员 → 作家 → 校对"""
    loop = AgentLoop(harness, llm)
    
    # Agent A: 研究
    child_h = harness.restrict({"allowed_tools": ["web_search", "read_file"], "max_steps": 10})
    research = AgentLoop(child_h, llm).run(f"研究以下主题并给出关键信息: {task}")
    
    # Agent B: 写作
    child_h = harness.restrict({"allowed_tools": ["write_file"], "max_steps": 15})
    draft = AgentLoop(child_h, llm).run(f"根据以下研究写一篇文章:\n{research}")
    
    # Agent C: 校对
    final = loop.run(f"校对并润色以下文章:\n{draft}")
    return final


# ===== 模式 2: Router =====
async def router(harness: Harness, llm, task: str):
    """调度 Agent 判断任务类型，分发给专业 Agent"""
    loop = AgentLoop(harness, llm)
    
    # 先让主 Agent 分类
    category = loop.run(
        f"将以下任务分类为: CODE / WRITING / RESEARCH / OTHER。只输出一个词。\n任务: {task}"
    ).strip()
    
    # 按类别分发
    tool_sets = {
        "CODE": ["read_file", "write_file", "run_shell"],
        "WRITING": ["write_file", "web_search"],
        "RESEARCH": ["web_search"],
        "OTHER": ["web_search"],
    }
    
    child_h = harness.restrict({"allowed_tools": tool_sets.get(category, []), "max_steps": 15})
    return AgentLoop(child_h, llm).run(task)


# ===== 模式 3: Debate/Review =====
async def debate(harness: Harness, llm, task: str, rounds: int = 3):
    """生成 → 评审 → 修改，反复迭代"""
    loop = AgentLoop(harness, llm)
    result = loop.run(task)
    
    for i in range(rounds):
        # 评审
        review = loop.run(f"严格评审以下输出，列出 3 个具体改进点:\n{result}")
        if "无改进点" in review or "无需修改" in review:
            break
        # 修改
        result = loop.run(f"根据以下评审意见修改:\n{review}\n\n原文:\n{result}")
    
    return result


# ===== 模式 4: Collaborative Graph =====
import asyncio

async def collaborative_graph(harness: Harness, llm, task: str):
    """规划者拆解任务 → 多个执行者并行 → 汇总"""
    loop = AgentLoop(harness, llm)
    
    # Step 1: 规划者拆解任务
    plan = loop.run(f"将以下任务拆解为 3-5 个独立的子任务，每行一个:\n{task}")
    subtasks = [s.strip() for s in plan.split("\n") if s.strip() and len(s) > 5]
    
    # Step 2: 并行执行
    async def run_subtask(sub_task):
        child_h = harness.restrict({"max_steps": 10})
        return AgentLoop(child_h, llm).run(sub_task)
    
    results = await asyncio.gather(*[run_subtask(st) for st in subtasks])
    
    # Step 3: 汇总
    combined = "\n\n---\n\n".join(
        f"子任务 {i+1}: {st}\n结果: {r}"
        for i, (st, r) in enumerate(zip(subtasks, results))
    )
    return loop.run(f"将以下子任务结果汇总成一份完整报告:\n{combined}")
```
